import sys
import collections
from pathlib import Path

ROOT = Path(sys.argv[1])
sys.path.insert(0, str(ROOT / "scripts/memory"))
import _lessons
import yaml

mods = {m["id"]: m for m in yaml.safe_load((ROOT / "modules.yaml").read_bytes().decode("utf-8"))["modules"]}
by = collections.defaultdict(list)
for p in _lessons.iter_lessons(ROOT / "docs/claude/memory"):
    fm = _lessons.read_lesson(p)
    for m in fm.get("module", []) if fm else []:
        if m in mods:
            by[m].append((p.stem, " ".join(fm["description"])))
tot = heads = 0
big = {}
for mid in sorted(by):
    ps = mods[mid].get("paths") or []
    if not ps:
        continue
    pp = [(x + "**") if x.endswith("/") else x for x in ps]
    head = (
        "---\npaths:\n"
        + "".join(f'  - "{x}"\n' for x in pp)
        + "---\n"
        + f"generated by scripts/memory/rules.py; edit lessons, not this file; more: search.py --module {mid}\n\n"
    )
    body = "".join(f"- {n} — {d}\n" for n, d in sorted(by[mid]))
    text = head + body
    tot += len(text)
    heads += len(head)
    if mid in ("process_module", "logger_module", "tools/backend_ctl"):
        big[mid] = len(text)
print("nodes", len(by), "total", tot, "heads", heads, big)
