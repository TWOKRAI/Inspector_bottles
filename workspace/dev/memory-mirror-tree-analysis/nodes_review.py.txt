import sys
import subprocess
import collections
from pathlib import Path

ROOT = Path(sys.argv[1])
sys.path.insert(0, str(ROOT / "scripts/memory"))
import _lessons
import yaml

mods = yaml.safe_load((ROOT / "modules.yaml").read_bytes().decode("utf-8"))["modules"]
ids = {m["id"]: m for m in mods}
print("modules:", len(mods))
lessons = _lessons.iter_lessons(ROOT / "docs/claude/memory")
print("lessons (iter_lessons):", len(lessons))
by = collections.defaultdict(list)
with_mod = without = unknown = 0
unk = collections.Counter()
for p in lessons:
    fm = _lessons.read_lesson(p)
    ms = fm.get("module", []) if fm else []
    if ms:
        with_mod += 1
    else:
        without += 1
    for m in ms:
        if m in ids:
            by[m].append((p.stem, " ".join(fm["description"])))
        else:
            unk[m] += 1
print("lessons with module:", with_mod, "without:", without, "unknown tags:", dict(unk))
print("modules with >=1 lesson:", len(by))
files = set(
    subprocess.run(["git", "-C", str(ROOT), "ls-files", "-z"], capture_output=True).stdout.decode("utf-8").split("\0")
)


def node(mid):
    ps = ids[mid].get("paths") or []
    pp = [(x + "**") if x.endswith("/") else x for x in ps]
    head = "---\npaths:\n" + "".join(f'  - "{x}"\n' for x in pp) + "---\n"
    head += f"<!-- generated by scripts/memory/rules.py, do not edit -->\nLessons for module {mid}. Body: docs/claude/memory/<name>.md. More: python scripts/memory/search.py --module {mid}\n\n"
    body = "".join(f"- {n} — {d}\n" for n, d in sorted(by[mid]))
    return head + body, body


tot_full = tot_lines_nl = tot_lines_nonl = 0
rows = []
for mid in sorted(by):
    full, body = node(mid)
    lines = body.splitlines()
    tot_full += len(full)
    tot_lines_nl += len(body)
    tot_lines_nonl += sum(len(l) for l in lines)
    rows.append((len(lines), len(body), mid))
rows.sort(reverse=True)
print("top nodes (lines, chars body):", rows[:6])
print("nodes >40:", sum(r[0] > 40 for r in rows), ">25:", sum(r[0] > 25 for r in rows))
print(
    "sum body chars with \n:",
    tot_lines_nl,
    " without \n:",
    tot_lines_nonl,
    " full node text (DESIGN 2 format):",
    tot_full,
)
print("full node bytes utf8:", sum(len(node(m)[0].encode()) for m in by))
# path problems
for mid in sorted(by):
    ps = ids[mid].get("paths") or []
    if not ps:
        print("NO-PATHS with lessons:", mid, len(by[mid]))
    for x in ps:
        if any(c in x for c in "*?["):
            print("GLOB in paths:", mid, x)
        if x.endswith("/"):
            if not any(f.startswith(x) for f in files):
                print("DEAD dir:", mid, x)
        else:
            if x not in files:
                isdir = any(f.startswith(x + "/") for f in files)
                print("NOT-A-FILE:", mid, x, "(is a dir w/o slash)" if isdir else "(absent)")
# nesting: nodes whose paths are under other nodes' paths
for a in sorted(by):
    for b in sorted(by):
        if a != b:
            for pa in ids[a].get("paths") or []:
                for pb in ids[b].get("paths") or []:
                    if pa.endswith("/") and pb.startswith(pa) and pb != pa:
                        print("NESTED:", b, pb, "under", a, pa)
