_NODE = re.compile(r"^(?:FAILED|ERROR) \S+::(?:\w+::)?(\w+)", re.M)


def run(inj_id: str) -> bool:
    path, old, new, expect = INJ[inj_id]
    original = path.read_bytes()
    text = original.decode("utf-8")
    nl = "\r\n" if "\r\n" in text else "\n"
    old_n, new_n = old.replace("\n", nl), new.replace("\n", nl)
    if text.count(old_n) != 1:
        print(f"[{inj_id}] АНКЕР НЕ НАЙДЕН РОВНО 1 РАЗ (найдено {text.count(old_n)}) — пропуск")
        return False
    try:
        path.write_bytes(text.replace(old_n, new_n).encode("utf-8"))
        env = dict(os.environ, PYTHONPATH=str(ROOT), PYTHONUTF8="1", QT_QPA_PLATFORM="offscreen")
        proc = subprocess.run(
            [sys.executable, "-m", "pytest", "-q", "-rfE", "--tb=no", "-p", "no:cacheprovider", *TARGETS],
            cwd=MF,
            env=env,
            capture_output=True,
            text=True,
            encoding="utf-8",
            errors="replace",
            timeout=600,
        )
    finally:
        path.write_bytes(original)
    out = proc.stdout + proc.stderr
    got = set(_NODE.findall(out))
    last = out.strip().splitlines()[-1] if out.strip() else "<нет вывода>"
    ok = expect <= got and not (got - expect)
    extra = sorted(got - expect)
    print(
        f"[{inj_id}] {'СОВПАЛО' if ok else 'РАСХОЖДЕНИЕ'} expect={sorted(expect)} got={sorted(got)} extra={extra} | {last}",
        flush=True,
    )
    return ok


if __name__ == "__main__":
    ids = sys.argv[1:] or list(INJ)
    results = {i: run(i) for i in ids}
    print(f"ИТОГ: совпало {sum(results.values())}/{len(results)}")
