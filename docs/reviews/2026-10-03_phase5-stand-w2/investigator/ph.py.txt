import json, sys, statistics as st

d = json.load(open(sys.argv[1], encoding="utf-8"))
tr = d["trace_proc"]
print("aff_E", d.get("aff_E"), "aff_P", d.get("aff_P"))
for lo, hi, nm in ((0, 10, "free"), (11, 20, "E-only"), (21, 31, "P-only")):
    rows = [r for r in tr if lo <= r[0] < hi and r[2] is not None]
    if len(rows) < 2:
        continue
    rate = (rows[-1][1] - rows[0][1]) / (rows[-1][0] - rows[0][0])
    print(
        f"{nm:7s} n={len(rows)} cycle_ms med={st.median(r[2] for r in rows):.2f} qw last={rows[-1][3]} med={st.median(r[3] for r in rows):.2f} frames/s={rate:.1f}"
    )
