def tests():
    out = []
    for pat in ("process_module/tests/test_t53_*.py", "router_module/tests/test_t53_*.py",
                "process_module/tests/test_t47d*.py", "router_module/tests/test_t47d*.py"):
        out += [str(p.relative_to(MOD)).replace("\\", "/") for p in sorted(MOD.glob(pat))]
    return out


def run():
    r = subprocess.run(
        [PY, "-m", "pytest", *tests(), "-q", "--tb=no", "-p", "no:cacheprovider", "-rfE"],
        cwd=MOD, env=env, capture_output=True, text=True, encoding="utf-8", errors="replace", timeout=600,
    )
    fails = sorted(set(re.findall(r"(?:FAILED|ERROR) (\S+)::(\S+)", r.stdout)))
    summary = [ln for ln in r.stdout.splitlines() if " passed" in ln or " failed" in ln or " error" in ln][-1:]
    by_file = {}
    for f, t in fails:
        by_file.setdefault(f.split("/")[-1], []).append(t)
    return summary, {k: len(v) for k, v in by_file.items()}, [t for _, t in fails][:12]


for name in sys.argv[1:]:
    patches = INJ.get(name, [])
    for f, a, b in patches:
        p = W / f
        with open(p, encoding="utf-8", newline="") as fh:
            s = fh.read()
        nl = "\r\n" if "\r\n" in s else "\n"
        a2, b2 = a.replace("\n", nl), b.replace("\n", nl)
        assert s.count(a2) == 1, (name, f, s.count(a2))
        with open(p, "w", encoding="utf-8", newline="") as fh:
            fh.write(s.replace(a2, b2))
    try:
        print(name, *run(), flush=True)
    finally:
        for f in {f for f, *_ in patches}:
            subprocess.run(["git", "checkout", "--", f], cwd=W)
