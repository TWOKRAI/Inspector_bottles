def run():
    failed = set()
    collected = 0
    for cwd, tests in ((ROOT / "multiprocess_framework/modules", FW_TESTS), (ROOT, ROOT_TESTS)):
        r = subprocess.run(
            [PY, "-m", "pytest", *tests, "-q", "-p", "no:cacheprovider", "-rf", "--tb=no"],
            cwd=cwd,
            env=ENV,
            capture_output=True,
            text=True,
            encoding="utf-8",
            errors="replace",
            timeout=600,
        )
        out = r.stdout
        failed |= {m.group(1) for m in re.finditer(r"^FAILED (\S+)", out, re.M)}
        m = re.search(r"(\d+) passed", out)
        f = re.search(r"(\d+) failed", out)
        collected += (int(m.group(1)) if m else 0) + (int(f.group(1)) if f else 0)
    return collected, sorted(failed)


def main():
    res = {}
    base = run()
    res["BASE"] = {"collected": base[0], "failed": base[1]}
    print("BASE", base[0], len(base[1]), flush=True)
    for name, (path, a, b) in INJ.items():
        p = ROOT / path
        s = p.read_text(encoding="utf-8")
        if s.count(a) != 1:
            res[name] = {"error": f"anchor count {s.count(a)}"}
            print(name, "ANCHOR", s.count(a), flush=True)
            continue
        p.write_text(s.replace(a, b), encoding="utf-8")
        try:
            col, fl = run()
        finally:
            subprocess.run(["git", "checkout", "--", path], cwd=ROOT)
        res[name] = {"pred": PRED[name], "collected": col, "failed": fl}
        print(name, col, len(fl), flush=True)
    out = Path(sys.argv[1])
    out.write_text(json.dumps(res, ensure_ascii=False, indent=1), encoding="utf-8")


if __name__ == "__main__":
    main()
