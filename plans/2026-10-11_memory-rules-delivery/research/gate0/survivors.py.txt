import json
import sys

sys.path.insert(0, "scripts/memory")
import _lessons
from pathlib import Path

res = json.load(open(sys.argv[1], encoding="utf-8"))
n = 0
for r in res:
    names = [r["new"]] + [o["old"] for o in r["olds"]]
    tags = {}
    for nm in names:
        for p in (
            Path("docs/claude/memory") / f"feedback_{nm}.md",
            Path("docs/claude/memory/_archive") / f"feedback_{nm}.md",
        ):
            if p.exists():
                fm = _lessons.read_lesson(p)
                tags[nm] = (p.parent.name, fm.get("module", []), fm.get("mechanism", []))
    anymod = any(t[1] for t in tags.values())
    n += anymod
    print(("M " if anymod else "- ") + r["new"], tags)
print("pairs with any module tag:", n)
