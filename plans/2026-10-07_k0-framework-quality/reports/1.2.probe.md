## probe.py N=20 (debounce, пауза truncate→write в секундах)
```python
import json, os, sys, tempfile, time
from pathlib import Path
from multiprocess_framework.modules.config_module.core.config import Config
from multiprocess_framework.modules.config_module.tools.watcher import ConfigFileWatcher

def run(mode, debounce, gap):
    d = Path(tempfile.mkdtemp()); f = d / "c.json"
    f.write_text(json.dumps({"v": "old"}))
    cfg = Config(initial_data={"v": "old"}); errs = []
    w = ConfigFileWatcher(f, cfg, debounce_seconds=debounce, log_error=errs.append); w.start()
    time.sleep(0.4)
    payload = json.dumps({"v": "final"})
    if mode == "write_text":
        f.write_text(payload)
    elif mode == "open_w_gap":      # truncate, пауза, запись
        h = open(f, "w"); h.flush(); time.sleep(gap); h.write(payload); h.close()
    elif mode == "atomic":
        t = f.with_name("c.json.tmp"); t.write_text(payload); os.replace(t, f)
    dl = time.monotonic() + 3.0
    while cfg.get("v") != "final" and time.monotonic() < dl: time.sleep(0.02)
    ok = cfg.get("v") == "final"; w.stop(); return ok, len(errs)

N = int(sys.argv[1])
for mode, deb, gap in [("write_text",1.0,0),("write_text",0.05,0),("open_w_gap",1.0,0.05),("open_w_gap",0.05,0.2),("atomic",1.0,0)]:
    res = [run(mode, deb, gap) for _ in range(N)]
    print(f"{mode:11s} debounce={deb:<4} gap={gap:<4} lost={sum(not r[0] for r in res)}/{N} errs_reported={sum(r[1] for r in res)}")
```

## probe2.py N=10 (YAML — формат боевых конфигов; debounce=1.0 — умолчание обоих боевых watcher)
```python
import json, sys, tempfile, time
from pathlib import Path
from multiprocess_framework.modules.config_module.core.config import Config
from multiprocess_framework.modules.config_module.tools.watcher import ConfigFileWatcher
from multiprocess_framework.modules.data_schema_module.serialization.converter import DataConverter

def run(kind):
    d = Path(tempfile.mkdtemp()); ext = ".yaml" if kind.startswith("yaml") else ".json"; f = d/("c"+ext)
    f.write_text('{"v": "old"}' if ext==".json" else "v: old\n")
    cfg = Config(initial_data={"v":"old"}); errs=[]
    w = ConfigFileWatcher(f, cfg, debounce_seconds=1.0, log_error=errs.append); w.start(); time.sleep(0.4)
    if kind == "save_to_file_json": DataConverter.save_to_file({"v":"final"}, f)
    elif kind == "yaml_slow":
        h=open(f,"w"); h.flush(); time.sleep(0.05); h.write("v: final\n"); h.close()
    elif kind == "yaml_chunked":   # два ключа, первый кусок сам по себе валидный YAML
        h=open(f,"w"); h.write("a: 1\n"); h.flush(); time.sleep(0.05); h.write("v: final\n"); h.close()
    dl=time.monotonic()+2.0
    while cfg.get("v")!="final" and time.monotonic()<dl: time.sleep(0.02)
    ok = cfg.get("v")=="final"; w.stop(); return ok, len(errs)
N=int(sys.argv[1])
for k in ("save_to_file_json","yaml_slow","yaml_chunked"):
    r=[run(k) for _ in range(N)]
    print(f"{k:18s} lost={sum(not x[0] for x in r)}/{N} errs_reported={sum(x[1] for x in r)}")
```
