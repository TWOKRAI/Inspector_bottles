### П.1. `multiprocess_framework/modules/process_module/generic/inbox_watch.py` (новый)

Docstring модуля lite (Purpose, Public API, `Stability: lite`, Pre/Post/Invariants из кода); `__all__ = ["INBOX_STALL_KEY", "InboxWatch", "parse_stall_s"]`.

```python
import math
import time
from collections.abc import Callable

INBOX_STALL_KEY = "inbox_stall_s"
_MIN_S, _MAX_S = 1.0, 600.0


def parse_stall_s(raw: object) -> tuple[float | None, str | None]:
    """(порог, текст для WARNING). None и 0 — выключено молча; негодное — выключено с текстом."""
    if raw is None or (type(raw) in (int, float) and raw == 0):
        return None, None
    if type(raw) not in (int, float) or not math.isfinite(raw) or not (_MIN_S <= raw <= _MAX_S):
        return None, f"{INBOX_STALL_KEY}={raw!r} негоден (число {_MIN_S}–{_MAX_S} с или 0) — предупреждение выключено"
    return float(raw), None


class InboxWatch:
    """Pre: observe() зовёт один поток (снимок готовности). Post: [] или ровно одна причина out_of_spec.
    Invariant: причина только при running, известных cycles/depth, depth > 0 и cycles без изменений >= stall_s."""

    def __init__(self, process_name: str, stall_s: float, clock: Callable[[], float] = time.monotonic) -> None:
        self._stall_s = stall_s
        self._clock = clock
        self._since: float | None = None
        self._last_cycles: int | None = None
        self._reason = {
            "node": process_name,
            "severity": "out_of_spec",
            "reason": f"{process_name}: кадры не принимаются",
            "source": "inbox",
        }

    def observe(self, running: bool, cycles: int | None, depth: int | None) -> list[dict]:
        if not running or cycles is None or depth is None:
            self._since = self._last_cycles = None
            return []
        now = self._clock()
        moved = cycles != self._last_cycles
        self._last_cycles = cycles
        if moved or depth <= 0:
            self._since = now if depth > 0 else None
            return []
        if self._since is None:
            self._since = now
            return []
        return [dict(self._reason)] if now - self._since >= self._stall_s else []
```
