def answer_of(run_idx: int, qid: str):

# ---------------------------------------------------------------- V0 и проверка «читаю score.py как есть»


def v0_verdict(q, ans, questions=None):
    return score.classify(q, ans, PREFIXES)


def check_v0_matches_record() -> None:
    for i, r in enumerate(rec["repeats"]):
        recorded = [row["verdict"] for row in r["score"]["per_question"]]
        mine = [v0_verdict(q, answer_of(i, q["id"])) for q in QUESTIONS]
        assert recorded == mine, (i, recorded, mine)


# ---------------------------------------------------------------- вспомогательное


def locs_of(ans) -> list[tuple[str, int | None, int | None]]:
    if ans is None:
        return []
    return [score.parse_location_range(loc, PREFIXES) for loc in ans["locations"]]


def entry_status(exp: dict, ans, tol: int | None = None) -> str:
    """Статус одной записи эталона для таблицы причин."""
    want = score.normalize_path(exp["path"])
    t = exp.get("tolerance", score.TOLERANCE_LINES) if tol is None else tol
    same = [(s, e) for (p, s, e) in locs_of(ans) if p == want]
    if not same:
        return "файл не назван"
    line = exp.get("line")
    if line is None:
        return "ok"
    ok = any(s is not None and s <= line + t and e >= line - t for s, e in same)
    if ok:
        return "ok"
    if all(s is None for s, _ in same):
        return "файл без строки"
    shown = ",".join(f"{s}" if s == e else f"{s}-{e}" for s, e in same if s is not None)
    return f"строка {shown} вне [{line - t},{line + t}]"


# ---------------------------------------------------------------- V1: файл-уровень


def v1_hit(q, ans) -> bool:
    expected = q["expected"]
    named = {p for (p, _, _) in locs_of(ans)}
    paths = [score.normalize_path(e["path"]) for e in expected]
    ok = [p in named for p in paths]
    return all(ok) if q.get("require", "any") == "all" else any(ok)


# ---------------------------------------------------------------- V2: частичный зачёт


def symbol_in_text(symbol: str | None, ans) -> bool:
    return bool(symbol and ans and re.search(r"(?<!\w)" + re.escape(symbol) + r"(?!\w)", ans["answer"], re.I))


def v2_question_score(q, ans, *, tol: int = 10, use_symbol: bool = False, path_only: float = 0.0) -> float:
    expected = q["expected"]
    total = 0.0
    for exp in expected:
        st = entry_status(exp, ans, tol)
        if st == "ok":
            total += 1.0
        elif st != "файл не назван":  # файл назван, строка не в окне
            if use_symbol and symbol_in_text(exp.get("symbol"), ans):
                total += 1.0
            else:
                total += path_only
    return total / len(expected)

def edited_questions(file_only: bool = False, groups: tuple[str, ...] | None = None) -> list[dict]:
    """groups: ограничить правки вопросами (None - все)."""
    edits = {**EDITS, **(EDITS_FILE_ONLY if file_only else {})}
    qs = copy.deepcopy(QUESTIONS)
    for q in qs:
        if groups is not None and q["id"] not in groups:
            continue
        for exp in q["expected"]:
            patch = edits.get((q["id"], exp.get("symbol")))
            if patch:
                exp.update(patch)
        if q["id"] in ADD:
            q["expected"].append(copy.deepcopy(ADD[q["id"]]))
    return qs


def dump_sets(outdir: Path) -> None:
    for name, fo in (("r1.V3.json", False), ("r1.V3b.json", True)):
        data = copy.deepcopy(qset)
        data["questions"] = edited_questions(fo)
        (outdir / name).write_text(json.dumps(data, ensure_ascii=False, indent=2) + "\n", encoding="utf-8")


# ---------------------------------------------------------------- вывод


def verdict_line(name: str, per_run: list[int], false=None) -> str:
    mn = min(per_run)
    k1 = "Kill-1 по hits: " + (">=7 (условие hits выполнено)" if mn >= 7 else f"<7 (hits.min {mn})")
    k2 = "Kill-2: " + ("зелёная" if mn >= 7 else "жёлтая" if mn == 6 else "СТОП")
    extra = f"  false={false} max={max(false)}" if false is not None else ""
    return f"{name:<34} hits={per_run} hits.min={mn} | {k1} | {k2}{extra}"


def table() -> None:
    names = ["V0"]
    print("Таблица 8x3: адреса агента | статусы записей эталона | вердикт score.py (V0)")
    for q in QUESTIONS:
        print(f"\n## {q['id']}  require={q.get('require')}  expected=" + "; ".join(
            f"{e['path']}:{e.get('line')}±{e.get('tolerance', 3)}" for e in q["expected"]))
        for i in range(3):
            a = answer_of(i, q["id"])
            sts = [entry_status(e, a) for e in q["expected"]]
            print(f"  r{i+1}: {v0_verdict(q, a):<5} conf={(a or {}).get('confidence')} locs={(a or {}).get('locations')}")
            for e, st in zip(q["expected"], sts):
                print(f"        {e['path'].split('/')[-1]}:{e.get('line')} -> {st}")

        ff = [sum(score.classify(q, answer_of(i, q["id"]), PREFIXES) == "false" for q in qs) for i in range(3)]
